import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from scipy.cluster.hierarchy import linkage, dendrogram

X = load_iris().data
X = StandardScaler().fit_transform(X)

wcss = []
for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(X)
    wcss.append(model.inertia_)

plt.plot(range(1, 11), wcss, marker='o')
plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.show()

dendrogram(linkage(X, method="ward"))
plt.title("Dendrogram")
plt.xlabel("Data Points")
plt.ylabel("Distance")
plt.show()

k = int(input("Enter number of clusters: "))

km = KMeans(n_clusters=k, random_state=42, n_init=10)
kmeans_labels = km.fit_predict(X)

hc = AgglomerativeClustering(n_clusters=k, linkage="ward")
hier_labels = hc.fit_predict(X)

print("\nK-Means Clusters:")
print(kmeans_labels)
print("Cluster sizes:", __import__('numpy').unique(kmeans_labels, return_counts=True)[1])

print("\nHierarchical Clusters:")
print(hier_labels)
print("Cluster sizes:", __import__('numpy').unique(hier_labels, return_counts=True)[1])

plt.scatter(X[:, 0], X[:, 1], c=kmeans_labels)
plt.title("K-Means Clustering")
plt.show()

plt.scatter(X[:, 0], X[:, 1], c=hier_labels)
plt.title("Hierarchical Clustering")
plt.show()
