
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

void sort(int a[], int n, int asc) {
    int i, j, t;

    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if ((asc && a[j] > a[j+1]) ||
                (!asc && a[j] < a[j+1])) {
                t = a[j];
                a[j] = a[j+1];
                a[j+1] = t;
            }
}

int main() {
    int a[100], n, i;
    pid_t pid;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if (n < 1 || n > 100) return 1;

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    pid = fork();

    if (pid == 0) {
        sort(a, n, 1);
        printf("Child (Ascending): ");
    }
    else if (pid > 0) {
        wait(NULL);
        sort(a, n, 0);
        printf("Parent (Descending): ");
    }
    else {
        printf("Fork failed!\n");
        return 1;
    }

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}
