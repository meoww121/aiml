
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define N 5

sem_t forks[N], room;

void *philosopher(void *arg) {
    int i = *(int *)arg;
    int right = (i + 1) % N;

    printf("Philosopher %d is thinking\n", i + 1);

    sem_wait(&room);
    sem_wait(&forks[i]);
    sem_wait(&forks[right]);

    printf("Philosopher %d is eating\n", i + 1);
    sleep(1);

    sem_post(&forks[right]);
    sem_post(&forks[i]);
    sem_post(&room);

    printf("Philosopher %d finished eating\n", i + 1);

    return NULL;
}

int main() {
    pthread_t t[N];
    int id[N];

    sem_init(&room, 0, N - 1);

    for (int i = 0; i < N; i++)
        sem_init(&forks[i], 0, 1);

    for (int i = 0; i < N; i++) {
        id[i] = i;
        pthread_create(&t[i], NULL, philosopher, &id[i]);
    }

    for (int i = 0; i < N; i++)
        pthread_join(t[i], NULL);

    for (int i = 0; i < N; i++)
        sem_destroy(&forks[i]);

    sem_destroy(&room);

    return 0;
}
